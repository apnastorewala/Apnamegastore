import React, { useState } from 'react';
import { 
  X, 
  MapPin, 
  User, 
  Phone, 
  Clock, 
  CreditCard, 
  Loader2,
  ShoppingBag
} from 'lucide-react';
import { CartItem, CheckoutFormData, StoreConfig, CompletedOrder } from '../types/grocery';
import { DELIVERY_SLOTS } from '../data/sampleProducts';
import { sendOrderToTelegramBot } from '../utils/telegramBot';

interface CheckoutModalProps {
  isOpen: boolean;
  onClose: () => void;
  items: CartItem[];
  storeConfig: StoreConfig;
  onOrderComplete: (order: CompletedOrder) => void;
}

export const CheckoutModal: React.FC<CheckoutModalProps> = ({
  isOpen,
  onClose,
  items,
  storeConfig,
  onOrderComplete,
}) => {
  const [formData, setFormData] = useState<CheckoutFormData>({
    fullName: '',
    phone: '',
    address: '',
    landmark: '',
    pincode: '',
    deliverySlot: DELIVERY_SLOTS[0].label,
    paymentMethod: 'cod',
    orderNotes: '',
  });

  const [errors, setErrors] = useState<Record<string, string>>({});
  const [isSending, setIsSending] = useState(false);

  if (!isOpen) return null;

  const totalMrp = items.reduce((sum, item) => sum + item.product.mrp * item.quantity, 0);
  const subtotal = items.reduce((sum, item) => sum + item.product.price * item.quantity, 0);
  const totalSavings = totalMrp - subtotal;
  const isFreeDelivery = subtotal >= storeConfig.freeDeliveryThreshold;
  const deliveryFee = isFreeDelivery ? 0 : storeConfig.deliveryFee;
  const totalPayable = subtotal + deliveryFee;

  const validate = (): boolean => {
    const newErrors: Record<string, string> = {};
    if (!formData.fullName.trim()) newErrors.fullName = 'Please enter your full name';
    const cleanPhone = formData.phone.replace(/\D/g, '');
    if (!cleanPhone || cleanPhone.length < 10) newErrors.phone = 'Enter valid 10-digit mobile number';
    if (!formData.address.trim()) newErrors.address = 'Full delivery address is required';
    if (!formData.landmark.trim()) newErrors.landmark = 'Nearby landmark is required for home delivery';
    if (!formData.pincode.trim() || formData.pincode.trim().length < 6) newErrors.pincode = 'Enter 6-digit pin code';
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const generateOrderMessage = (orderId: string, timestamp: string) => {
    const paymentLabel = 
      formData.paymentMethod === 'cod' 
        ? 'Cash on Delivery (COD)' 
        : formData.paymentMethod === 'upi_doorstep' 
          ? 'UPI Scanner at Doorstep' 
          : 'Online UPI Advance';

    const itemsText = items
      .map((item, idx) => `${idx + 1}. ${item.product.name} (${item.product.packSize}) - ₹${item.product.price * item.quantity} (Qty: ${item.quantity})`)
      .join('\n');

    return (
`🛒 NEW MART HOME DELIVERY ORDER!
-----------------------------------------
Items:
${itemsText}

Subtotal: ₹${subtotal}
Delivery: ${deliveryFee === 0 ? 'FREE' : `₹${deliveryFee}`}
Total Bill: ₹${totalPayable} (Saved: ₹${totalSavings})
-----------------------------------------
Customer Name: ${formData.fullName.trim()}
Phone: ${formData.phone.trim()}
Delivery Address: ${formData.address.trim()}
Nearby Landmark: ${formData.landmark.trim()}
Pin Code: ${formData.pincode.trim()}
Preferred Slot: ${formData.deliverySlot}
Payment Mode: ${paymentLabel}
${formData.orderNotes ? `Special Notes: ${formData.orderNotes.trim()}\n` : ''}-----------------------------------------
Order ID: #${orderId}
Store: ${storeConfig.storeName}
Date & Time: ${timestamp}`
    );
  };

  const handleConfirmOrder = async () => {
    if (!validate()) return;
    setIsSending(true);

    const orderId = `MART-${Math.floor(1000 + Math.random() * 9000)}`;
    const formattedDate = new Date().toLocaleDateString('en-IN', {
      day: 'numeric',
      month: 'short',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
    });

    const telegramMessage = generateOrderMessage(orderId, formattedDate);

    const completedOrder: CompletedOrder = {
      orderId,
      date: formattedDate,
      items,
      customer: formData,
      subtotal,
      discount: totalSavings,
      deliveryFee,
      totalPayable,
      totalSavings,
      telegramMessage,
      botStatus: 'manual',
    };

    if (storeConfig.telegramBotToken?.trim() && storeConfig.telegramChatId?.trim()) {
      try {
        const botResult = await sendOrderToTelegramBot(completedOrder, storeConfig);
        completedOrder.botStatus = botResult.success ? 'sent_via_bot' : 'manual';
      } catch {
        completedOrder.botStatus = 'manual';
      }
    }

    setIsSending(false);
    onOrderComplete(completedOrder);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-black/60 backdrop-blur-xs overflow-y-auto">
      <div className="bg-white rounded-xl shadow-2xl max-w-2xl w-full my-6 overflow-hidden flex flex-col max-h-[92vh]">
        {/* Header */}
        <div className="px-5 py-4 bg-[#2874F0] text-white flex items-center justify-between">
          <div>
            <h2 className="text-base sm:text-lg font-bold flex items-center gap-2">
              <MapPin className="w-5 h-5 text-yellow-300" />
              Home Delivery Checkout
            </h2>
            <p className="text-xs text-blue-100 mt-0.5">
              Enter your address to receive order directly at your doorstep
            </p>
          </div>
          <button onClick={onClose} className="p-1 rounded-full text-white/80 hover:text-white hover:bg-white/20 transition">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Form Body */}
        <div className="p-5 overflow-y-auto space-y-5">
          <div className="bg-blue-50 border border-blue-200/80 rounded-lg p-3.5 flex flex-wrap items-center justify-between gap-2 text-xs">
            <div>
              <span className="text-slate-600">Cart Total: </span>
              <strong className="text-slate-900 font-bold">{items.length} Items</strong>
              <span className="mx-2 text-slate-300">|</span>
              <span className="text-slate-600">Total Bill: </span>
              <strong className="text-slate-950 font-mono text-sm">₹{totalPayable}</strong>
            </div>
            {totalSavings > 0 && (
              <span className="font-bold text-emerald-700 bg-emerald-100/90 px-2 py-0.5 rounded">
                Saving ₹{totalSavings} 🎉
              </span>
            )}
          </div>

          {/* Customer Personal Details */}
          <div className="space-y-3">
            <h3 className="text-xs font-bold uppercase tracking-wider text-slate-700 flex items-center gap-1.5">
              <User className="w-4 h-4 text-[#2874F0]" />
              <span>1. Customer Information</span>
            </h3>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  Full Customer Name <span className="text-red-500">*</span>
                </label>
                <input
                  type="text"
                  value={formData.fullName}
                  onChange={(e) => setFormData({ ...formData, fullName: e.target.value })}
                  placeholder="e.g. Rahul Verma"
                  className={`w-full px-3 py-2 text-xs sm:text-sm rounded border ${
                    errors.fullName ? 'border-red-500 bg-red-50/30' : 'border-slate-300'
                  } focus:outline-none focus:ring-2 focus:ring-[#2874F0]`}
                />
                {errors.fullName && <p className="text-[11px] text-red-600 mt-0.5">{errors.fullName}</p>}
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  10-Digit Mobile Number <span className="text-red-500">*</span>
                </label>
                <div className="relative">
                  <span className="absolute inset-y-0 left-0 pl-3 flex items-center text-slate-400 text-xs font-medium">+91</span>
                  <input
                    type="tel"
                    maxLength={10}
                    value={formData.phone}
                    onChange={(e) => setFormData({ ...formData, phone: e.target.value.replace(/\D/g, '') })}
                    placeholder="9876543210"
                    className={`w-full pl-11 pr-3 py-2 text-xs sm:text-sm rounded border ${
                      errors.phone ? 'border-red-500 bg-red-50/30' : 'border-slate-300'
                    } focus:outline-none focus:ring-2 focus:ring-[#2874F0] font-mono`}
                  />
                </div>
                {errors.phone && <p className="text-[11px] text-red-600 mt-0.5">{errors.phone}</p>}
              </div>
            </div>
          </div>

          {/* Delivery Address & Landmark */}
          <div className="space-y-3 pt-2 border-t border-slate-100">
            <h3 className="text-xs font-bold uppercase tracking-wider text-slate-700 flex items-center gap-1.5">
              <MapPin className="w-4 h-4 text-[#2874F0]" />
              <span>2. Delivery Address & Landmark</span>
            </h3>

            <div>
              <label className="block text-xs font-semibold text-slate-700 mb-1">
                Full House/Flat No., Building & Street Address <span className="text-red-500">*</span>
              </label>
              <textarea
                rows={2}
                value={formData.address}
                onChange={(e) => setFormData({ ...formData, address: e.target.value })}
                placeholder="e.g. Flat 302, Sai Residency, Near Old Market Road"
                className={`w-full px-3 py-2 text-xs sm:text-sm rounded border ${
                  errors.address ? 'border-red-500 bg-red-50/30' : 'border-slate-300'
                } focus:outline-none focus:ring-2 focus:ring-[#2874F0]`}
              />
              {errors.address && <p className="text-[11px] text-red-600 mt-0.5">{errors.address}</p>}
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  Nearby Landmark <span className="text-red-500">*</span>
                </label>
                <input
                  type="text"
                  value={formData.landmark}
                  onChange={(e) => setFormData({ ...formData, landmark: e.target.value })}
                  placeholder="e.g. Near Shiv Mandir / Sweets Shop"
                  className={`w-full px-3 py-2 text-xs sm:text-sm rounded border ${
                    errors.landmark ? 'border-red-500 bg-red-50/30' : 'border-slate-300'
                  } focus:outline-none focus:ring-2 focus:ring-[#2874F0]`}
                />
                {errors.landmark && <p className="text-[11px] text-red-600 mt-0.5">{errors.landmark}</p>}
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1">
                  Pin Code <span className="text-red-500">*</span>
                </label>
                <input
                  type="text"
                  maxLength={6}
                  value={formData.pincode}
                  onChange={(e) => setFormData({ ...formData, pincode: e.target.value.replace(/\D/g, '') })}
                  placeholder="e.g. 110001"
                  className={`w-full px-3 py-2 text-xs sm:text-sm rounded border ${
                    errors.pincode ? 'border-red-500 bg-red-50/30' : 'border-slate-300'
                  } focus:outline-none focus:ring-2 focus:ring-[#2874F0] font-mono`}
                />
                {errors.pincode && <p className="text-[11px] text-red-600 mt-0.5">{errors.pincode}</p>}
              </div>
            </div>
          </div>

          {/* Preferred Delivery Slot */}
          <div className="space-y-2 pt-2 border-t border-slate-100">
            <h3 className="text-xs font-bold uppercase tracking-wider text-slate-700 flex items-center gap-1.5">
              <Clock className="w-4 h-4 text-[#2874F0]" />
              <span>3. Preferred Delivery Time Slot</span>
            </h3>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
              {DELIVERY_SLOTS.map((slot) => {
                const isChecked = formData.deliverySlot === slot.label;
                return (
                  <label
                    key={slot.id}
                    className={`flex items-start gap-2.5 p-2.5 rounded-lg border text-xs cursor-pointer transition ${
                      isChecked
                        ? 'border-[#2874F0] bg-blue-50/70 ring-1 ring-[#2874F0]'
                        : 'border-slate-200 hover:bg-slate-50'
                    }`}
                  >
                    <input
                      type="radio"
                      name="deliverySlot"
                      checked={isChecked}
                      onChange={() => setFormData({ ...formData, deliverySlot: slot.label })}
                      className="mt-0.5 text-[#2874F0]"
                    />
                    <div>
                      <span className="font-bold text-slate-900">{slot.label}</span>
                      <span className="text-[11px] text-slate-500 block mt-0.5">{slot.timeRange}</span>
                    </div>
                  </label>
                );
              })}
            </div>
          </div>

          {/* Payment Method */}
          <div className="space-y-2 pt-2 border-t border-slate-100">
            <h3 className="text-xs font-bold uppercase tracking-wider text-slate-700 flex items-center gap-1.5">
              <CreditCard className="w-4 h-4 text-[#2874F0]" />
              <span>4. Payment Mode</span>
            </h3>

            <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 text-xs">
              <label className={`p-2.5 rounded-lg border cursor-pointer ${formData.paymentMethod === 'cod' ? 'border-[#2874F0] bg-blue-50/70 ring-1 ring-[#2874F0]' : 'border-slate-200'}`}>
                <div className="flex items-center gap-2">
                  <input type="radio" name="paymentMethod" value="cod" checked={formData.paymentMethod === 'cod'} onChange={() => setFormData({ ...formData, paymentMethod: 'cod' })} />
                  <span className="font-bold text-slate-900">Cash on Delivery</span>
                </div>
                <span className="text-[11px] text-slate-500 mt-1 pl-5 block">Pay cash at doorstep</span>
              </label>

              <label className={`p-2.5 rounded-lg border cursor-pointer ${formData.paymentMethod === 'upi_doorstep' ? 'border-[#2874F0] bg-blue-50/70 ring-1 ring-[#2874F0]' : 'border-slate-200'}`}>
                <div className="flex items-center gap-2">
                  <input type="radio" name="paymentMethod" value="upi_doorstep" checked={formData.paymentMethod === 'upi_doorstep'} onChange={() => setFormData({ ...formData, paymentMethod: 'upi_doorstep' })} />
                  <span className="font-bold text-slate-900">UPI Scanner</span>
                </div>
                <span className="text-[11px] text-slate-500 mt-1 pl-5 block">Scan QR code at door</span>
              </label>

              <label className={`p-2.5 rounded-lg border cursor-pointer ${formData.paymentMethod === 'upi_online' ? 'border-[#2874F0] bg-blue-50/70 ring-1 ring-[#2874F0]' : 'border-slate-200'}`}>
                <div className="flex items-center gap-2">
                  <input type="radio" name="paymentMethod" value="upi_online" checked={formData.paymentMethod === 'upi_online'} onChange={() => setFormData({ ...formData, paymentMethod: 'upi_online' })} />
                  <span className="font-bold text-slate-900">Online UPI</span>
                </div>
                <span className="text-[11px] text-slate-500 mt-1 pl-5 block">Pay advance via UPI</span>
              </label>
            </div>
          </div>
        </div>

        {/* Modal Action Footer: SINGLE BUTTON 'Confirm Order' */}
        <div className="p-4 bg-slate-50 border-t border-slate-200 flex items-center justify-between gap-4">
          <div className="text-xs text-slate-600">
            <span className="text-slate-500">Payable Amount:</span>{' '}
            <strong className="text-base font-extrabold text-slate-950 font-mono">₹{totalPayable}</strong>
          </div>

          <button
            type="button"
            onClick={handleConfirmOrder}
            disabled={isSending}
            className="flex-1 sm:flex-none sm:min-w-[220px] bg-[#fb641b] hover:bg-[#f45207] text-white font-extrabold text-sm py-3 px-6 rounded-lg shadow-md transition active:scale-98 flex items-center justify-center gap-2 disabled:opacity-75 cursor-pointer"
          >
            {isSending ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                <span>Placing Order...</span>
              </>
            ) : (
              <>
                <ShoppingBag className="w-4 h-4" />
                <span>Confirm Order</span>
              </>
            )}
          </button>
        </div>
      </div>
    </div>
  );
};
