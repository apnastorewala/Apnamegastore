export function formatOrderForTelegramHtml(
  order: CompletedOrder,
  storeConfig: StoreConfig
): string {
  const itemsText = order.items
    .map((item, idx) => {
      const itemTotal = item.product.price * item.quantity;
      return `${idx + 1}. <b>${escapeHtml(item.product.name)}</b> (${escapeHtml(item.product.packSize)})\n   └ ₹${item.product.price} × ${item.quantity} = <b>₹${itemTotal}</b>`;
    })
    .join('\n');

  const paymentLabel =
    order.customer.paymentMethod === 'cod'
      ? '💵 Cash on Delivery (COD)'
      : order.customer.paymentMethod === 'upi_doorstep'
        ? '📱 UPI Scanner at Doorstep (GPay / PhonePe / Paytm)'
        : '💳 Online UPI Advance';

  return (
`🛒 <b>NEW MART HOME DELIVERY ORDER!</b>
━━━━━━━━━━━━━━━━━━━━━
<b>Order ID:</b> #${escapeHtml(order.orderId)}
<b>Date & Time:</b> ${escapeHtml(order.date)}
<b>Store:</b> ${escapeHtml(storeConfig.storeName)}

📦 <b>ORDERED ITEMS:</b>
${itemsText}

💰 <b>BILL SUMMARY:</b>
Subtotal: ₹${order.subtotal}
Delivery Charges: ${order.deliveryFee === 0 ? '<b>FREE</b> 🚀' : `₹${order.deliveryFee}`}
<b>Total Payable: ₹${order.totalPayable}</b>
${order.totalSavings > 0 ? `🎉 <i>Customer Saved: ₹${order.totalSavings}</i>\n` : ''}
━━━━━━━━━━━━━━━━━━━━━
👤 <b>CUSTOMER DETAILS:</b>
• <b>Name:</b> ${escapeHtml(order.customer.fullName)}
• <b>Mobile:</b> <code>+91 ${escapeHtml(order.customer.phone)}</code>
• <b>Address:</b> ${escapeHtml(order.customer.address)}
• <b>Landmark:</b> 📍 <b>${escapeHtml(order.customer.landmark)}</b>
• <b>Pin Code:</b> ${escapeHtml(order.customer.pincode)}
• <b>Preferred Slot:</b> ${escapeHtml(order.customer.deliverySlot)}
• <b>Payment Mode:</b> ${paymentLabel}
${order.customer.orderNotes ? `• <b>Notes:</b> <i>${escapeHtml(order.customer.orderNotes)}</i>\n` : ''}━━━━━━━━━━━━━━━━━━━━━
⚡ <i>Dispatch delivery boy with fresh packed items.</i>`
  );
}

function escapeHtml(text: string): string {
  if (!text) return '';
  return text
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;');
}

export async function sendOrderToTelegramBot(
  order: CompletedOrder,
  storeConfig: StoreConfig
): Promise<{ success: boolean; error?: string; messageId?: number }> {
  const token = storeConfig.telegramBotToken?.trim();
  const chatId = storeConfig.telegramChatId?.trim();

  if (!token || !chatId) {
    return {
      success: false,
      error: 'Bot Token or Chat ID not configured. Please add them in src/config.ts',
    };
  }

  const cleanToken = token.startsWith('bot') ? token.slice(3) : token;
  const url = `https://api.telegram.org/bot${cleanToken}/sendMessage`;
  const messageHtml = formatOrderForTelegramHtml(order, storeConfig);

  try {
    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        chat_id: chatId,
        text: messageHtml,
        parse_mode: 'HTML',
      }),
    });

    const data = await response.json();

    if (data.ok) {
      return { success: true, messageId: data.result?.message_id };
    } else {
      return {
        success: false,
        error: data.description || 'Failed to send message via Telegram Bot API',
      };
    }
  } catch (err: any) {
    return {
      success: false,
      error: err?.message || 'Network error connecting to Telegram Bot API',
    };
  }
}

  onClose,

  onUpdateQuantity,
  onRemoveItem,
  onProceedToCheckout,

}) => {

  const totalPayable = subtotal + deliveryFee;

  const amountNeededForFreeDelivery = Math.max(0, storeConfig.freeDeliveryThreshold - subtotal);
  const freeDeliveryProgress = Math.min(100, Math.round((subtotal / storeConfig.freeDeliveryThreshold) * 100));

  return (
    <div className="fixed inset-0 z-50 overflow-hidden">
      <div onClick={onClose} className="fixed inset-0 bg-black/50 backdrop-blur-xs transition-opacity duration-300" />

      <div className="fixed inset-y-0 right-0 max-w-full flex pl-10">
        <div className="w-screen max-w-md bg-white shadow-2xl flex flex-col">
          {/* Header */}
          <div className="px-5 py-4 bg-[#2874F0] text-white flex items-center justify-between shadow-sm">
            <div className="flex items-center gap-2">
              <ShoppingBag className="w-5 h-5 text-yellow-300" />
              <h2 className="font-bold text-base tracking-tight">
                My Mart Cart ({totalItemsCount} {totalItemsCount === 1 ? 'item' : 'items'})
              </h2>
            </div>
            <button onClick={onClose} className="p-1 rounded-full text-white/80 hover:text-white hover:bg-white/20 transition">
              <X className="w-5 h-5" />
            </button>
          </div>

          {/* Free Delivery Bar */}
          {items.length > 0 && (
            <div className="bg-blue-50 px-5 py-2.5 border-b border-blue-100 text-xs">
              {isFreeDelivery ? (
                <div className="flex items-center gap-1.5 text-emerald-700 font-bold">
                  <Sparkles className="w-4 h-4 fill-emerald-500 text-emerald-500" />
                  <span>Yay! You unlocked FREE Home Delivery for this order! 🚀</span>
                </div>
              ) : (
                <div>
                  <div className="flex items-center justify-between text-slate-700 font-medium mb-1">
                    <span>Add <strong>₹{amountNeededForFreeDelivery}</strong> more for <strong>FREE Delivery</strong></span>
                    <span className="font-mono text-[11px] text-blue-700 font-bold">{freeDeliveryProgress}%</span>
                  </div>
                  <div className="w-full bg-blue-200 h-1.5 rounded-full overflow-hidden">
                    <div className="bg-emerald-500 h-full transition-all duration-300" style={{ width: `${freeDeliveryProgress}%` }} />
                  </div>
                </div>
              )}
            </div>
          )}

          {/* Cart Item List */}
          <div className="flex-1 overflow-y-auto p-5 divide-y divide-slate-100">
            {items.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-center p-6 text-slate-500">
                <ShoppingBag className="w-16 h-16 text-[#2874F0] mb-3 opacity-60" />
                <h3 className="font-bold text-slate-800 text-lg mb-1">Your Cart is Empty</h3>
                <p className="text-xs text-slate-400 mb-5">Add pantry essentials, atta, oil & snacks to get fast delivery.</p>
                <button onClick={onContinueShopping} className="px-5 py-2.5 bg-[#2874F0] hover:bg-blue-700 text-white text-xs font-bold rounded-lg transition">
                  Browse Mart Products
                </button>
              </div>
            ) : (
              items.map((item) => (
                <div key={item.product.id} className="py-3 flex gap-3 items-center">
                  <img src={item.product.image} alt={item.product.name} className="w-16 h-16 object-contain p-1 border border-slate-100 rounded bg-slate-50" />
                  <div className="flex-1">
                    <h4 className="text-xs font-bold text-slate-800 line-clamp-1">{item.product.name}</h4>
                    <span className="text-[11px] text-slate-500">{item.product.packSize}</span>
                    <div className="flex items-baseline gap-2 mt-1">
                      <span className="font-mono font-bold text-slate-900 text-sm">₹{item.product.price}</span>
                      <span className="text-xs text-slate-400 line-through">₹{item.product.mrp}</span>
                    </div>
                  </div>

                  <div className="flex items-center gap-1 border border-slate-200 rounded px-1.5 py-0.5 bg-slate-50">
                    <button onClick={() => onUpdateQuantity(item.product.id, item.quantity - 1)} className="p-0.5 text-slate-600 hover:text-black">
                      <Minus className="w-3 h-3" />
                    </button>
                    <span className="font-bold font-mono text-xs px-1.5">{item.quantity}</span>
                    <button onClick={() => onUpdateQuantity(item.product.id, item.quantity + 1)} className="p-0.5 text-[#2874F0]">
                      <Plus className="w-3 h-3" />
                    </button>
                  </div>

                  <button onClick={() => onRemoveItem(item.product.id)} className="text-slate-400 hover:text-red-500 p-1">
                    <Trash2 className="w-4 h-4" />
                  </button>
                </div>
              ))
            )}
          </div>

          {/* Drawer Footer */}
          {items.length > 0 && (
            <div className="p-5 bg-slate-50 border-t border-slate-200 space-y-3">
              <div className="space-y-1.5 text-xs text-slate-600">
                <div className="flex justify-between">
                  <span>Subtotal:</span>
                  <span className="font-mono">₹{subtotal}</span>
                </div>
                <div className="flex justify-between">
                  <span>Delivery Charges:</span>
                  <span className="font-semibold text-emerald-700">{deliveryFee === 0 ? 'FREE' : `₹${deliveryFee}`}</span>
                </div>
                <div className="flex justify-between text-base font-extrabold text-slate-900 pt-2 border-t border-slate-200">
                  <span>Total Amount:</span>
                  <span className="font-mono">₹{totalPayable}</span>
                </div>
              </div>

              <button
                onClick={onProceedToCheckout}
                className="w-full bg-[#fb641b] hover:bg-[#f45207] text-white font-extrabold text-sm py-3 px-4 rounded-lg shadow-md transition flex items-center justify-center gap-2 cursor-pointer"
              >
                <span>Proceed to Checkout</span>
                <ArrowRight className="w-4 h-4" />
              </button>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};
